function doGet(e) {

/* ============================================================
   LOGARE ERORI IN COL. C
============================================================ */

function logErrorToSheet(from, msg) {
  const sh = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME);
  const data = sh.getDataRange().getValues();

  for (let i = 1; i < data.length; i++) {
    let nr = data[i][1] ? data[i][1].toString() : "";
    if (nr === from.toString()) {
      sh.getRange(i+1, 3).setValue(msg);
      return;
    }
  }

  sh.appendRow(["NECUNOSCUT", from, msg]);
}


/* ============================================================
   HANDLER PRINCIPAL (FULL FLOW)
============================================================ */

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const entry = data.entry?.[0]?.changes?.[0]?.value;
    const messageObj = entry?.messages?.[0];

    if (!messageObj)
      return ContentService.createTextOutput("NO MESSAGE");

    const from = messageObj.from;
    const type = messageObj.type;
    const text = messageObj.text?.body ?? "";

    logErrorToSheet(from, "Mesaj primit, tip: " + type);


    /* ============================================================
       1️⃣ BUTOANE DIN TEMPLATE ("Prețuri")
    ============================================================= */

    if (messageObj.button) {
      const replyId = messageObj.button.text || messageObj.button.payload;
      logErrorToSheet(from, "Buton TEMPLATE apăsat: " + replyId);

      if (replyId === "Prețuri") {
        sendMessage(from, "📍 În ce *zonă* locuiți?");
        saveTemp(from, "stage", "zona");
        return ContentService.createTextOutput("OK");
      }

      logErrorToSheet(from, "Buton TEMPLATE necunoscut: " + replyId);
      return ContentService.createTextOutput("OK");
    }


    /* ============================================================
       2️⃣ BUTOANE INTERACTIVE
    ============================================================= */

    if (type === "interactive") {
      const replyId = messageObj.interactive?.button_reply?.id ||
                      messageObj.interactive?.list_reply?.id;

      logErrorToSheet(from, "Buton INTERACTIV apăsat: " + replyId);

      if (!replyId) {
        logErrorToSheet(from, "Eroare: replyId lipsă");
        return ContentService.createTextOutput("OK");
      }

      /* ---------- A. ALEGERE POMPA ---------- */
      if (replyId === "pompa_da" || replyId === "pompa_nu") {
        const rasp = replyId === "pompa_da" ? "DA" : "NU";
        saveTemp(from, "pompa", rasp);

        sendClassList(from);
        saveTemp(from, "stage", "clasa");
        return ContentService.createTextOutput("OK");
      }

      /* ---------- B. ALEGERE CLASĂ BETON ----------- */
      if (replyId.startsWith("beton_")) {
        const clasa = replyId.replace("beton_", "");
        saveTemp(from, "clasa", clasa);

        processFinalCalculation(from);
        return ContentService.createTextOutput("OK");
      }

      /* ---------- C. BUTON FINAL "CĂTRE AGENT" ---------- */
      if (replyId === "catre_agent_final") {
        handleSendToAgent(from);
        return ContentService.createTextOutput("OK");
      }

      logErrorToSheet(from, "Buton INTERACTIV necunoscut: " + replyId);
      return ContentService.createTextOutput("OK");
    }


    /* ============================================================
       3️⃣ FLOW TEXT PROGRESIV
    ============================================================= */

    const stage = getTemp(from, "stage");
    logErrorToSheet(from, "Stage actual: " + stage);


    /* ---------- PAS 1: ZONA ---------- */
    if (stage === "zona") {
      saveTemp(from, "zona", text);
      sendMessage(from, "🛣 Care este *strada exactă* a proiectului?");
      saveTemp(from, "stage", "strada");
      return ContentService.createTextOutput("OK");
    }

    /* ---------- PAS 2: STRADA ---------- */
    if (stage === "strada") {
      saveTemp(from, "strada", text);
      sendMessage(from, "📦 De câți metri cubi aveți nevoie?");
      saveTemp(from, "stage", "cantitate");
      return ContentService.createTextOutput("OK");
    }

    /* ---------- PAS 3: CANTITATE ---------- */
    if (stage === "cantitate") {
      const mc = parseFloat(text.replace(",", "."));

      if (isNaN(mc) || mc <= 0) {
        sendMessage(from, "⚠️ Introduceți o valoare validă (ex: 12)");
        return ContentService.createTextOutput("OK");
      }

      saveTemp(from, "cant", mc);

      sendPompaButtons(from);
      saveTemp(from, "stage", "pompa");
      return ContentService.createTextOutput("OK");
    }

  } catch (err) {
    logErrorToSheet("ERR", "Eroare generala: " + err);
  }

  return ContentService.createTextOutput("OK");
}

/* ============================================================
   CALC DISTANTA ORS
============================================================ */

function getDistanceKM(startAddress, endAddress) {
  try {
    const startRes = UrlFetchApp.fetch(
      `https://api.openrouteservice.org/geocode/search?api_key=${ORS_API_KEY}&text=${encodeURIComponent(startAddress)}`
    );
    const startCoords = JSON.parse(startRes).features[0].geometry.coordinates;

    const endRes = UrlFetchApp.fetch(
      `https://api.openrouteservice.org/geocode/search?api_key=${ORS_API_KEY}&text=${encodeURIComponent(endAddress)}`
    );
    const endCoords = JSON.parse(endRes).features[0].geometry.coordinates;

    const routeRes = UrlFetchApp.fetch(
      `https://api.openrouteservice.org/v2/directions/driving-car?api_key=${ORS_API_KEY}`,
      {
        method: "post",
        contentType: "application/json",
        payload: JSON.stringify({ coordinates: [startCoords, endCoords] })
      }
    );

    const meters = JSON.parse(routeRes).routes[0].summary.distance;
    return parseFloat((meters / 1000).toFixed(1));

  } catch (err) {
    return -1;
  }
}


/* ============================================================
   CALCUL FINAL — MESAJ CLIENT
============================================================ */

function processFinalCalculation(phone) {

  const zona = getTemp(phone, "zona");
  const strada = getTemp(phone, "strada");
  const cant = parseFloat(getTemp(phone, "cant"));
  const pompa = getTemp(phone, "pompa");
  const clasa = getTemp(phone, "clasa");

  const preturiBeton = {
    "C 8/10": 315,
    "C 12/15": 325,
    "C 16/20": 335,
    "C 20/25": 350,
    "C 25/30": 370,
    "C 30/37": 390
  };

  const pretMC = preturiBeton[clasa];
  const costBeton = cant * pretMC;

  let costPompa = 0;
  if (pompa === "DA") {
    costPompa = Math.max(700, cant * 25);
  }

  const fullAddress = zona + ", " + strada;
  const kmOneWay = getDistanceKM(STATION_ADDRESS, fullAddress);

  if (kmOneWay < 0) {
    sendMessage(phone, "❗ Eroare la calcularea distanței. Reîncercați.");
    return;
  }

  const drumuri = Math.ceil(cant / 9);

  // ❗ NOUA FORMULĂ TRANSPORT
  const kmTotal = kmOneWay * 2 * drumuri;
  const kmFacturati = Math.max(0, kmTotal - 20);
  const costTransport = kmFacturati * 15;

  const totalFaraTVA = costBeton + costPompa + costTransport;
const totalCuTVA = Math.round(totalFaraTVA * 1.21);

const msg =
  "💰 TOTAL cu TVA inclus: " + totalCuTVA + " lei\n\n" +
  "Trimitem datele către un agent?";

  sendMessageWithButton(phone, msg, "Către agent", "catre_agent_final");
}
